# Python dependencies for the offline HOCBF gain-learning pipeline.
# Target: Python 3.10.12
#
#   pip install -r requirements.txt
#
# Verified against every import in src/. No ROS 2 and no CasADi are needed:
# all reported results come from the offline closed-loop simulator.
#
# Pinned to the versions used to produce the published tables. The solver stack
# (osqp / clarabel / scs) and scipy are pinned as well, since QP-solver and
# BLAS/LAPACK versions can slightly shift the reported numbers.

# --- direct imports ---
# numpy is dictated by Pinocchio: pin==3.9.0 requires cmeel-boost~=1.89.0,
# whose wheels require numpy>=2.2,<2.3 on Python 3.10 and numpy>=2.3,<2.4 on
# Python >= 3.11. (numpy 1.26.4 + pin==3.9.0 is not installable via pip; the
# original 3.10.12 machine likely had Pinocchio from apt/ROS instead.)
numpy          >= 2.2, < 2.3 ; python_version <  "3.11"
numpy          >= 2.3, < 2.4 ; python_version >= "3.11"
pandas         == 2.3.3
matplotlib     == 3.10.9
torch          == 2.11.0
cvxpy          == 1.7.5
cvxpylayers    == 0.1.9
pin            == 3.9.0     # Pinocchio. Imported as "import pinocchio".
                            # If the wheel fails: conda install -c conda-forge pinocchio
cmeel-urdfdom  == 4.0.1     # pin 3.9.0's binaries link liburdfdom_*.so.4 and
cmeel-tinyxml2 == 10.0.0    # libtinyxml2.so.10, but its metadata allows newer
                            # majors, which break "import pinocchio"
pyyaml         == 6.0.3     # imported as "import yaml"

# --- solver / numerical stack pulled in by cvxpy (pinned for reproducibility) ---
osqp           == 1.1.1     # the QP solver used at evaluation time
clarabel       == 0.11.1
scs            == 3.2.11
scipy          == 1.15.3
